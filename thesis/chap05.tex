\chapter{Perfect matching of bipartite graph}

\section{Hall proof}

\subsubsection{Definition: Neighbourhood function}
Let $G = (A, B, E)$ be a bipartite graph. \newline
Neighbourhood function N is defined as:
\[
    \forall S \subseteq A: N(S) \equiv \{b \in B | \exists a \in S: (a, b) \in E \}
\]

\subsection{Hall proof existence theorem}

Let $G = (A, B, E)$ be a bipartite graph with a perfect matching $P$.
Let $N$ be the neighbourhood function in $G$.
Let $E_{NPM}$ be set of all perfectly unmatchable edges in $G$.

\begin{gather*}
    \forall (a, b) \in E_{NPM} \exists S \subseteq A: \\
    a \notin S \\
    b \in N(S) \\
    N(S) = P(S) \\
\end{gather*}

\subsubsection{Proof of 5.1.1}

Let $(a, b) \in E_{NPM}$. \newline
Let $A' \equiv A \setminus \{a\}$,
$B' \equiv B \setminus \{b\}$ and
$E' = \{ (u, v) \in E \mid u \neq a \land v \neq b \}$. \newline
Let $G' \equiv (A', B', E')$. \newline
Let $N'$ be the neighbourhood function in $G'$.

\paragraph{Lemma 1} $\forall S' \subseteq A': \left| N(S') \right| \geq \left| S' \right|$

\paragraph{Proof of lemma 1}
$G$ has perfect matching, therefore according to Hall's theorem:
\[
    \forall S \subseteq A: \left| N(S) \right| \geq \left| S \right|
\]
From definition of $A'$:
\[
    A' \equiv A \setminus \{a\} \implies \forall S' \subseteq A': S' \subseteq A
\]

End of proof of lemma 1.

\paragraph{Lemma 2}
$\forall S' \subseteq A': N'(S') = N(S') \setminus \{b\}$

\paragraph{Proof of lemma 2}

We prove each inequality by itself.

\begin{gather*}
    E' \equiv \{(a', b') \in E | a' \neq a \land b' \neq b\} \\
    \implies \\
    \forall (a', b') \in E': (a', b') \in E \\
    \implies \\
    \forall S' \subseteq A': \forall b' \in B: \forall a' \in S': (a', b') \in E' \implies (a', b') \in E \\
    \implies \\
    \forall S' \subseteq A': \forall b' \in B: (\exists a' \in S': (a', b') \in E') \implies (\exists a' \in S': (a', b') \in E) \\
    \implies \\
    \forall S' \subseteq A': N'(S') \subseteq N(S') \\
    \implies \\
    \forall S' \subseteq A': N'(S') \subseteq N(S') \setminus \{b\} \\
\end{gather*}

Fifth implication holds, because $N'(S') \subseteq B' \equiv B \setminus \{b\}$.

\begin{gather*}
    E' \equiv \{(a', b') \in E | a' \neq a \land b' \neq b\} \\
    \implies \\
    \forall (a', b') \in E: (a', b') \in E' \lor a' = a \lor b' = b \\
    \implies \\
    \forall S' \subseteq A': \forall b' \in B: \forall a' \in S': (a', b') \in E \implies (a', b') \in E' \lor a' = a \lor b' = b \\
    \implies \\
    \forall S' \subseteq A': \forall b' \in B: \forall a' \in S': (a', b') \in E \implies (a', b') \in E' \lor b' = b \\
    \implies \\
    \forall S' \subseteq A': \forall b' \in B: (\exists a' \in S': (a', b') \in E') \implies (\exists a' \in S': (a', b') \in E) \lor b' = b \\
    \implies \\
    \forall S' \subseteq A': \forall b' \in B: b' \in N(S') \implies b' \in N'(S') \lor b' = b \\
    \implies \\
    \forall S' \subseteq A': \forall b' \in B: b' \in N(S') \setminus \{b\} \implies b' \in N'(S') \\
    \implies \\
    \forall S' \subseteq A': N(S') \setminus \{b\} \subseteq N'(S') \\
\end{gather*}

Third implication holds, because $a' \in S' \subseteq A' \equiv A \setminus \{a\} \implies a' \neq a$. \newline
End of proof of lemma 2.

\paragraph{Lemma 3}
$G'$ does not have a perfect matching.

\paragraph{Proof of lemma 3}
Assume $G'$ has a perfect matching $P'$.
Then $P' \cup \{(a, b)\})$ is perfect matching for $G$.
That is a contradiction, because $(a, b)$ is perfectly unmatchable edge in $G$.
Therefore, $G'$ does not have a perfect matching. \newline
End of proof of lemma 3.

\paragraph{Lemma 4} $\exists S' \subseteq A': \left|N(S') \setminus \{b\}\right| < \left|S'\right|$

\paragraph{Proof of lemma 4}
$G'$ is not perfectly matchable (lemma 3).
According to Hall's theorem:
\[
    \exists S' \subseteq A': \left|N'(S')\right| < \left|S'\right|
\]
Using lemma 2 we get:
\[
    \exists S' \subseteq A': \left|N(S') \setminus \{b\}\right| < \left|S'\right|
\]
End of proof of lemma 4.

\bigskip

Now we put lemma 1 and lemma 4 together, which makes:
\[
    \exists S' \subseteq A': \left|N(S')\right| \geq \left|S'\right| > \left|N(S') \setminus \{b\} \right| \geq \left|N(S')\right| - 1
\]

Because all values are whole and first and last value differ by 1, there can only be one inequality:
\[
    \exists S' \subseteq A': \left|N(S')\right| = \left|S'\right| > \left|N(S') \setminus \{b\} \right| = \left|N(S')\right| - 1
\]

Because $A' \equiv A \setminus \{a\}$:
\begin{gather*}
    \exists S \subseteq A: \\
    a \notin S \\
    \left|N(S)\right| = \left|S\right| > \left|N(S) \setminus \{b\} \right| = \left|N(S)\right| - 1 \\
\end{gather*}

Now we can prove all three parts of 5.1.1:
\begin{gather*}
    a \notin S \\
    \left|N(S) \setminus \{b\} \right| = \left|N(S)\right| - 1 \implies b \in N(S) \\
    \left|N(S)\right| = \left|S\right| \implies N(S) = P(S) \\
\end{gather*}

End of proof of 5.1.1.
